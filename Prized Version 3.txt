local function IsAllowed(player)
    if not player then return false end
    return AllowedPlayers[player.Name] ~= nil
end

local function CleanupPlayer(player)
    if not player then return end
    local userId = player.UserId
    if PlayerData[userId] then
        local data = PlayerData[userId]

        -- Stop new dynamic systems first
        if data.StopLightningFlicker then
            pcall(function() data.StopLightningFlicker() end)
        end
        if data.StopGroundArcs then
            pcall(function() data.StopGroundArcs() end)
        end

        if data.stopColorUpdate then
            pcall(function() data.stopColorUpdate() end)
        end

        if data.connections then
            for _, conn in pairs(data.connections) do
                if conn and typeof(conn) == "RBXScriptConnection" and conn.Connected then
                    pcall(function() conn:Disconnect() end)
                end
            end
        end

        if data.cleanupObjects then
            for _, obj in pairs(data.cleanupObjects) do
                if obj and typeof(obj) == "Instance" then
                    pcall(function() obj:Destroy() end)
                end
            end
        end

        PlayerData[userId] = nil
        print("✓ Cleaned up " .. player.Name)
    end
end

local function createTrail(part, color, props, offsetAmount)
    if not part or not part.Parent then return end
    
    for i=1,2 do
        local success, err = pcall(function()
            local t = Instance.new("Trail")
            local a0 = Instance.new("Attachment", part)
            local a1 = Instance.new("Attachment", part)

            local offset = (i == 1) and Vector3.new(0, offsetAmount, 0) or Vector3.new(0, -offsetAmount, 0)
            a0.Position = offset
            a1.Position = offset * -1
            a0.Visible = false
            a1.Visible = false

            t.Attachment0 = a0
            t.Attachment1 = a1
            t.Enabled = false
            t.Texture = props.Texture
            t.TextureLength = props.TextureLength
            t.TextureMode = props.TextureMode
            t.FaceCamera = props.FaceCamera
            t.Lifetime = props.Lifetime
            t.WidthScale = props.WidthScale
            t.Transparency = props.Transparency
            if color then
                t.Color = ColorSequence.new(color)
            else
                t.Color = props.Color
            end
            t.Parent = part

            table.insert(allTrails, t)
            table.insert(playerData.cleanupObjects, t)
            table.insert(playerData.cleanupObjects, a0)
            table.insert(playerData.cleanupObjects, a1)
        end)
        
        if not success then
            warn("Failed to create trail for " .. player.Name .. ": " .. tostring(err))
        end
    end
end

local function SetupAllTrails()
    for _, t in pairs(allTrails) do
        if t and t.Parent then
            pcall(function()
                t:Destroy()
            end)
        end
    end
    allTrails = {}
    playerData.allTrails = allTrails
    
    if not CharacterRef or not CharacterRef.Parent then return end
    
    for _, partName in pairs(LIMBS) do
        local part = CharacterRef:FindFirstChild(partName)
        if part and part.Parent then

            local isArm = partName:find("Arm") or partName:find("Hand")
            local color = isArm and partColors.arms or partColors.legs
            createTrail(part, color, trailProps1, 0.2)
            createTrail(part, nil, trailProps2, 0.25)
        end
    end
    
    print("✓ Created " .. #allTrails .. " trails for " .. player.Name)
end

local noTrailParts = {
	["Torso"] = true, ["UpperTorso"] = true, ["LowerTorso"] = true,
	["HumanoidRootPart"] = true, ["Head"] = true
}

for _,v in pairs(CharacterRef:GetChildren()) do
	if v:IsA("BasePart") and not noTrailParts[v.Name] then
		local a0 = Instance.new("Attachment",v)
		local a1 = Instance.new("Attachment",v)

		-- DCEU-style: a1 fans outward from the character instead of trailing behind.
		-- Origin (a0) sits at the part center; endpoint (a1) is pushed far out
		-- horizontally so the trail slices AWAY from the body to the ground.
		local isLeft  = v.Name:find("Left")  ~= nil
		local isRight = v.Name:find("Right") ~= nil
		local spreadDist = math.random(35, 80) / 10  -- 3.5–8.0 studs outward
		local offsetY = -1.5  -- slight downward for ground-skimming look
		local offsetX, offsetZ

		if math.random() < 0.5 then
			-- Side-biased: follows which side the limb is on
			if isLeft then
				offsetX = -spreadDist
				offsetZ = (math.random() - 0.5) * 3
			elseif isRight then
				offsetX = spreadDist
				offsetZ = (math.random() - 0.5) * 3
			else
				-- Centre parts: random flat direction
				local ang = math.random() * math.pi * 2
				offsetX = math.cos(ang) * spreadDist
				offsetZ = math.sin(ang) * spreadDist
			end
		else
			-- 50 % chance: fully random horizontal direction (no up/down)
			local ang = math.random() * math.pi * 2
			offsetX = math.cos(ang) * spreadDist
			offsetZ = math.sin(ang) * spreadDist
		end

		a0.Position = Vector3.new(0, 0, 0)
		a1.Position = Vector3.new(offsetX, offsetY, offsetZ)

		local x = Instance.new("Trail")
		x.Attachment0 = a0
		x.Attachment1 = a1
		x.Brightness = 1
		x.FaceCamera = false
		x.LightEmission = 1

local CameraShakeState = {FadingIn=0,FadingOut=1,Sustained=2,Inactive=3}
local function newCameraShakeInstance(magnitude,roughness,fadeInTime,fadeOutTime)
	fadeInTime=fadeInTime or 0
	fadeOutTime=fadeOutTime or 0
	local inst={}
	inst.Magnitude=magnitude
	inst.Roughness=roughness
	inst.PositionInfluence=Vector3.new()
	inst.RotationInfluence=Vector3.new()
	inst.DeleteOnInactive=true
	inst.roughMod=1
	inst.magnMod=1
	inst.fadeOutDuration=fadeOutTime
	inst.fadeInDuration=fadeInTime
	inst.sustain=(fadeInTime>0)
	inst.currentFadeTime=(fadeInTime>0) and 0 or 1
	inst.tick=Random.new():NextNumber(-100,100)
	function inst:UpdateShake(dt)
		local tk=self.tick
		local ft=self.currentFadeTime
		if self.fadeInDuration>0 and self.sustain then
			if ft<1 then ft=ft+dt/self.fadeInDuration
			elseif self.fadeOutDuration>0 then self.sustain=false end
		end
		if not self.sustain then ft=ft-dt/self.fadeOutDuration end
		self.tick=self.sustain and (tk+dt*self.Roughness*self.roughMod) or (tk+dt*self.Roughness*self.roughMod*ft)
		self.currentFadeTime=ft
		return Vector3.new(math.noise(tk,0)*0.5,math.noise(0,tk)*0.5,math.noise(tk,tk)*0.5)*self.Magnitude*self.magnMod*ft
	end
	function inst:StartFadeOut(t) if t==0 then self.currentFadeTime=0 end self.fadeOutDuration=t self.fadeInDuration=0 self.sustain=false end
	function inst:StartFadeIn(t) if t==0 then self.currentFadeTime=1 end self.fadeInDuration=t or self.fadeInDuration self.fadeOutDuration=0 self.sustain=true end
	function inst:IsShaking() return self.currentFadeTime>0 or self.sustain end
	function inst:IsFadingOut() return (not self.sustain) and self.currentFadeTime>0 end
	function inst:IsFadingIn() return self.currentFadeTime<1 and self.sustain and self.fadeInDuration>0 end
	function inst:GetState()
		if self:IsFadingIn() then return CameraShakeState.FadingIn
		elseif self:IsFadingOut() then return CameraShakeState.FadingOut
		elseif self:IsShaking() then return CameraShakeState.Sustained
		else return CameraShakeState.Inactive end
	end
	return inst
end

local CurrentShaker=(function()
	local shaker={}
	shaker._running=false
	shaker._renderName="CameraShaker_"..player.Name
	shaker._renderPriority=Enum.RenderPriority.Camera.Value
	shaker._camShakeInstances={}
	shaker._removeInstances={}
	function shaker:Start()
		if self._running then return end
		self._running=true
		RunService:BindToRenderStep(self._renderName,self._renderPriority,function(dt)
			local instances=self._camShakeInstances
			local pos=Vector3.new()
			local rot=Vector3.new()
			for i=1,#instances do
				local inst=instances[i]
				local state=inst:GetState()
				if state==CameraShakeState.Inactive and inst.DeleteOnInactive then
					table.insert(self._removeInstances,i)
				elseif state~=CameraShakeState.Inactive then
					local offset=inst:UpdateShake(dt)
					pos=pos+offset*inst.PositionInfluence
					rot=rot+offset*inst.RotationInfluence
				end
			end
			for i=#self._removeInstances,1,-1 do
				table.remove(instances,self._removeInstances[i])
				self._removeInstances[i]=nil
			end
			cam.CFrame=cam.CFrame*CFrame.new(pos)*CFrame.Angles(0,math.rad(rot.Y),0)*CFrame.Angles(math.rad(rot.X),0,math.rad(rot.Z))
		end)
	end
	function shaker:Stop()
		if not self._running then return end
		RunService:UnbindFromRenderStep(self._renderName)
		self._running=false
	end
	function shaker:Shake(shakeInst)
		table.insert(self._camShakeInstances,shakeInst)
		return shakeInst
	end
	return shaker
end)()

-- ================================================================
-- FIX: HeartbeatUpdate no longer clone/destroys trails every frame
-- Original was ~960+ Instance operations/sec at 16 body parts × 60fps
-- Now it just toggles Enabled once per 2 frames for the flicker feel
-- Water detection and TODC are preserved exactly
-- ================================================================
local heartbeatFrame = 0

local function HeartbeatUpdate()
    heartbeatFrame += 1

    -- Lightweight trail "pop" flicker: toggle enabled state briefly
    -- This resets trail segments (same visual effect as the old clone/destroy)
    -- but costs nearly nothing compared to Clone() + Destroy()
    if heartbeatFrame % 2 == 0 then
        for Part, Data in pairs(TrailsPerPart) do
            local T1, T2 = Data[1], Data[2]
            -- Randomly pick one trail to snap-reset per frame
            local target = (math.random(1,2) == 1) and T1 or T2
            if target and target.Parent then
                -- Brief toggle clears the existing trail segment history
                target.Enabled = false
                target.Enabled = true
                -- Vary the lifetime for a non-uniform, organic look
                target.Lifetime = math.random(5, 22) / 100
            end
        end
    end

    -- Water detection (original behavior preserved)
    if heartbeatFrame % 30 == 0 then
        for _, Part in pairs(workspace:GetChildren()) do
            if Part.Name == "Water" and Part:IsA("BasePart") then
                CreateInvisibleWater(Part)
            end
        end
    end

    if _G.TODC then
        game:GetService('Lighting').TimeOfDay = "00:00:00"
    end
end

local function UpdateSounds()
    if not enabled then return end
    
    local currentSpeed = h.WalkSpeed
    local isMoving = h.MoveDirection.Magnitude > 0
    local isRunning = isMoving and currentSpeed > 0
    
    if isRunning then
        if not wasRunning then
            if not runSound1.IsPlaying then runSound1:Play() end
            if not runSound3.IsPlaying then runSound3:Play() end
            if not runSound4.IsPlaying then runSound4:Play() end
        end

        if not runSound2.IsPlaying then runSound2:Play() end
        if not runSound5.IsPlaying then runSound5:Play() end
    else
        if wasRunning or (not isMoving and currentSpeed == 0) then
            if runSound1.IsPlaying then runSound1:Stop() end
            if runSound2.IsPlaying then runSound2:Stop() end
            if runSound3.IsPlaying then runSound3:Stop() end
            if runSound4.IsPlaying then runSound4:Stop() end
            if runSound5.IsPlaying then runSound5:Stop() end
        end
    end
    
    if not isMoving and currentSpeed == 0 and wasRunning then
        if not stopSound.IsPlaying then
            stopSound:Play()
        end
    end
    
    wasRunning = isRunning
    lastWalkSpeed = currentSpeed
end

-- ================================================================
-- LIGHTNING SYSTEM 1: DYNAMIC TRAIL FLICKER ENGINE
--
-- Research basis: DCEU/Arrowverse Flash lightning is "barely contained
-- energy". Each limb group crackles at its own rhythm — legs are most
-- active (ground contact), arms flicker mid-rate, torso most stable.
-- The key is ASYMMETRY and IRREGULARITY. Real lightning never pulses
-- uniformly. It snaps on/off at different times across the body.
--
-- Implementation:
-- - Three independent flicker loops run on offset timers
-- - Each loop owns a "group" of body-part trails
-- - Lifetime varies per tick (shorter = crisper, longer = smoother)
-- - math.sin wave + random spike = organic non-repeating pattern
-- - When G_Optimized: slower intervals, higher on-chance (less spam)
-- ================================================================

local flickerActive = false
local flickerThread = nil

local function StartLightningFlicker()
    if flickerActive then return end
    flickerActive = true

    flickerThread = task.spawn(function()
        local tick_leg = 0
        local tick_arm = 7  -- offset so arms and legs don't sync
        local tick_all = 3  -- offset for allTrails

        while flickerActive do
            tick_leg += 1
            tick_arm += 1
            tick_all += 1

            local speed = (h and h.Parent and h.WalkSpeed) or 0
            -- intensity: 0.2 at idle (lightning still crackles slightly), 1.0 at full speed
            local intensity = math.clamp(speed / math.max(customSpeed, 1), 0.2, 1.0)

            -- === tlist: BASE body trails - the most visible layer ===
            -- These are the trails from the simple loop at the top of setup.
            -- They flicker fast and form the "electricity wrapping the body" look.
            for idx, trail in ipairs(tlist) do
                if trail and trail.Parent then
                    -- Sine wave base creates a slow pulse, random spikes create jitter
                    local sineWave = math.sin(tick_leg * 0.4 + idx * 0.9) * 0.25 + 0.75
                    local rand = math.random()
                    if rand < intensity * sineWave then
                        trail.Enabled = true
                        -- Vary lifetime: short = sharp bolt feel, long = glow trail feel
                        -- Mix both for that DCEU "energy crackling with spikes" look
                        if rand < 0.15 then
                            trail.Lifetime = math.random(5, 10) / 100   -- short crackle spike
                        else
                            trail.Lifetime = math.random(12, 28) / 100  -- normal trail
                        end
                    elseif rand < (0.08 * intensity) then
                        -- Snap-off: this is the "lightning bolt pops" effect
                        -- A small percentage of the time, a trail turns off completely
                        -- creating the visual of a bolt vanishing and reappearing
                        trail.Enabled = false
                    end
                    -- else: leave as-is (no unnecessary writes)
                end
            end

            -- === t trails: SECONDARY lightning layer ===
            -- These use the lightning texture and are slightly slower to flicker
            -- creating a layered depth effect (fast crackle on top of slower glow)
            if tick_arm % 2 == 0 then
                for idx, trail in ipairs(t) do
                    if trail and trail.Parent then
                        local sineArm = math.sin(tick_arm * 0.3 + idx * 1.1) * 0.2 + 0.8
                        local rand = math.random()
                        if rand < intensity * sineArm * 0.9 then
                            trail.Enabled = true
                            trail.Lifetime = math.random(15, 40) / 100
                        elseif rand < 0.05 * intensity then
                            trail.Enabled = false
                        end
                    end
                end
            end

            -- === allTrails: THIRD layer (from createTrail) ===
            -- Slowest flicker rate — these provide the "corona glow" around limbs
            if tick_all % 3 == 0 then
                for idx, trail in ipairs(allTrails) do
                    if trail and trail.Parent then
                        if math.random() < intensity * 0.85 then
                            trail.Enabled = true
                        elseif math.random() < 0.04 then
                            trail.Enabled = false
                        end
                    end
                end
            end

            -- Dynamic wait: G_Optimized uses longer intervals
            task.wait(_G.G_Optimized and (math.random(4, 8) / 100) or (math.random(2, 5) / 100))
        end
    end)
end

local function StopLightningFlicker()
    flickerActive = false
    if flickerThread then
        task.cancel(flickerThread)
        flickerThread = nil
    end
    -- Note: caller is responsible for setting trails to correct enabled/disabled state
end

playerData.StopLightningFlicker = StopLightningFlicker


-- ================================================================
-- LIGHTNING SYSTEM 2: DCEU GROUND ARC SYSTEM
--
-- Research basis:
-- DCEU Flash (Justice League, The Flash 2023): Lightning originates
-- FROM the speedster's core/torso and arcs DOWN to random ground
-- contact points. It doesn't just trail behind — it actively reaches
-- OUT and SNAPS to the ground like a Tesla coil discharge.
--
-- Unity/Unreal technique: Procedural noise-displaced line segments.
-- In Roblox equivalent: Beam instance with:
--   - High Segments count (10-24) = more control points = more jagged
--   - Random CurveSize0/CurveSize1 (positive AND negative values)
--     = S-curve deformation = zigzag lightning shape
--   - Very thin Width (0.03-0.15 studs) = realistic bolt thickness
--   - Very short Lifetime (30-180ms) = the "pop and vanish" feel
--   - ColorSequence: white core → speedster color → white tip
--
-- Raycast: fires from multiple body parts at slight downward angles
-- to find the ground. If it hits, places a temporary invisible anchor
-- part at the hit position, connects a Beam, then Debris cleans it up.
-- A secondary "branch" arc fires at 35% chance to simulate branching.
--
-- Bugs avoided:
-- - Position error: bodyAttach is parented to the part (not workspace)
--   so it moves with the character and needs no position updates
-- - groundAttach is at fixed world position (anchored part) so the
--   Beam naturally stretches as the character moves — looks correct
-- ================================================================

local groundArcActive = false
local groundArcThread = nil
local groundRayParams = RaycastParams.new()
groundRayParams.FilterType = Enum.RaycastFilterType.Exclude

-- Parts to arc FROM — prioritizes feet/legs (most visible ground contact)
-- then arms and hands, then torso as fallback
local GROUND_ARC_SOURCE_PARTS = {
    "LeftFoot", "RightFoot",
    "LeftLowerLeg", "RightLowerLeg",
    "Left Leg", "Right Leg",
    "LeftUpperLeg", "RightUpperLeg",
    "HumanoidRootPart",
    "UpperTorso", "LowerTorso", "Torso",
    "LeftHand", "RightHand",
    "LeftLowerArm", "RightLowerArm",
}

local Debris = game:GetService("Debris")

local function FireGroundArc(originPart, lightColor)
    if not originPart or not originPart.Parent then return end
    if not CharacterRef or not CharacterRef.Parent then return end
    if not HRP or not HRP.Parent then return end

    -- Update raycast exclusion list each call (character may have changed)
    groundRayParams.FilterDescendantsInstances = {CharacterRef}

    -- Variable spread: mostly downward, with horizontal deviation
    -- This makes lightning fan outward from feet rather than always going straight down
    local spreadX = (math.random() - 0.5) * 5
    local spreadZ = (math.random() - 0.5) * 5
    local rayLength = math.random(5, 20)
    local rayDir = Vector3.new(spreadX * 0.3, -rayLength, spreadZ * 0.3)

    local result = workspace:Raycast(originPart.Position, rayDir, groundRayParams)
    if not result then return end

    local groundPos = result.Position
    -- Slight random offset at ground contact = simulates lightning branching
    -- to a point slightly away from direct hit (like real ground strikes)
    groundPos = groundPos + Vector3.new(
        (math.random() - 0.5) * 2.5,
        0.02,  -- tiny upward offset so beam doesn't clip into floor
        (math.random() - 0.5) * 2.5
    )

    -- Temporary invisible anchor at ground position
    -- Anchored = true means it stays in world space while character moves
    -- This is intentional: the ground point is FIXED, the body end moves
    -- This creates the "lightning stretching as you run past" feel
    local groundPart = Instance.new("Part")
    groundPart.Anchored = true
    groundPart.CanCollide = false
    groundPart.CastShadow = false
    groundPart.Transparency = 1
    groundPart.Size = Vector3.new(0.05, 0.05, 0.05)
    groundPart.Position = groundPos
    groundPart.Parent = workspace

    local groundAttach = Instance.new("Attachment", groundPart)

    -- Body attachment with small random offset from part center
    -- Simulates lightning originating from different points on the limb
    local bodyAttach = Instance.new("Attachment", originPart)
    bodyAttach.Position = Vector3.new(
        (math.random() - 0.5) * 0.6,
        (math.random() - 0.5) * 0.5,
        (math.random() - 0.5) * 0.4
    )

    local whiteColor = Color3.new(1, 1, 1)

    -- === MAIN BEAM ARC ===
    local beam = Instance.new("Beam")
    beam.Attachment0 = bodyAttach   -- body end (moves with character)
    beam.Attachment1 = groundAttach  -- ground end (fixed)
    beam.FaceCamera = true
    beam.LightEmission = 1
    beam.LightInfluence = 0
    beam.Brightness = math.random(6, 14)
    -- Segments: more = more control points = more jagged zigzag path
    -- 10-24 gives good DCEU-style jaggedness without performance issues
    beam.Segments = math.random(10, 24)
    -- Width: very thin for realistic bolt appearance
    beam.Width0 = math.random(4, 14) / 100   -- origin end (body) slightly thicker
    beam.Width1 = math.random(1, 7) / 100    -- ground end thinner (tapers to point)
    -- CurveSize: THIS IS THE KEY TO JAGGEDNESS
    -- Positive + negative values on the two ends create an S-curve.
    -- Beam then interpolates across Segments creating the zigzag.
    -- Larger absolute values = more extreme zigzag.
    beam.CurveSize0 = (math.random() - 0.5) * 7
    beam.CurveSize1 = (math.random() - 0.5) * 7
    beam.Texture = "rbxassetid://16607923"  -- lightning bolt texture (from trailProps1)
    beam.TextureLength = math.random(1, 4)
    beam.TextureMode = Enum.TextureMode.Static
    -- Color: white core → speedster color → white tip (matches DCEU look)
    beam.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, whiteColor),
        ColorSequenceKeypoint.new(0.25, lightColor),
        ColorSequenceKeypoint.new(0.75, lightColor),
        ColorSequenceKeypoint.new(1, whiteColor)
    })
    -- Transparency: mostly opaque in middle, fades at ends
    beam.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.15),
        NumberSequenceKeypoint.new(0.4, 0),
        NumberSequenceKeypoint.new(0.6, 0),
        NumberSequenceKeypoint.new(1, 0.6)
    })
    beam.Parent = groundPart

    -- Lifetime: 30ms–180ms. The wide range gives unpredictable "pop" timing.
    -- Short bolts feel like sharp cracks; longer ones feel like sustained arcs.
    local lifetime = math.random(3, 18) / 100

    -- Schedule cleanup
    Debris:AddItem(groundPart, lifetime + 0.08)
    Debris:AddItem(bodyAttach, lifetime + 0.08)

    -- Fade-out tween: starts at 40% of lifetime, finishes by end
    -- This creates the "lightning dissipates" effect rather than hard-cutting
    task.delay(lifetime * 0.35, function()
        if beam and beam.Parent then
            ts:Create(beam, TweenInfo.new(lifetime * 0.65, Enum.EasingStyle.Expo, Enum.EasingDirection.Out), {
                Transparency = NumberSequence.new(1),
                Width0 = 0,
                Width1 = 0
            }):Play()
        end
    end)

    -- === BRANCH ARC (35% chance) ===
    -- Secondary thinner bolt that fires from same origin to a nearby ground point
    -- Simulates lightning branching — the single most recognizable DCEU Flash trait
    if math.random() < 0.35 and not _G.G_Optimized then
        task.delay(math.random(1, 3) / 100, function()  -- tiny delay so branch appears slightly after main
            if not CharacterRef or not CharacterRef.Parent then return end
            if not originPart or not originPart.Parent then return end

            local branchPart = Instance.new("Part")
            branchPart.Anchored = true
            branchPart.CanCollide = false
            branchPart.CastShadow = false
            branchPart.Transparency = 1
            branchPart.Size = Vector3.new(0.05, 0.05, 0.05)
            -- Branch lands at a different spot from the main arc
            branchPart.Position = groundPos + Vector3.new(
                (math.random() - 0.5) * 4,
                0.02,
                (math.random() - 0.5) * 4
            )
            branchPart.Parent = workspace

            local bAttach0 = Instance.new("Attachment", originPart)
            bAttach0.Position = bodyAttach.Position + Vector3.new(
                (math.random() - 0.5) * 0.4, 0, 0
            )
            local bAttach1 = Instance.new("Attachment", branchPart)

            local branchBeam = Instance.new("Beam")
            branchBeam.Attachment0 = bAttach0
            branchBeam.Attachment1 = bAttach1
            branchBeam.FaceCamera = true
            branchBeam.LightEmission = 1
            branchBeam.LightInfluence = 0
            branchBeam.Brightness = math.random(3, 8)
            branchBeam.Segments = math.random(6, 16)
            -- Branches are thinner than main arc
            branchBeam.Width0 = math.random(1, 6) / 100
            branchBeam.Width1 = math.random(1, 3) / 100
            branchBeam.CurveSize0 = (math.random() - 0.5) * 5
            branchBeam.CurveSize1 = (math.random() - 0.5) * 5
            branchBeam.Texture = "rbxassetid://16607923"
            branchBeam.TextureLength = 2
            branchBeam.TextureMode = Enum.TextureMode.Static
            branchBeam.Color = ColorSequence.new(lightColor, whiteColor)
            branchBeam.Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.4),
                NumberSequenceKeypoint.new(0.5, 0.1),
                NumberSequenceKeypoint.new(1, 1)
            })
            branchBeam.Parent = branchPart

            local bLifetime = lifetime * 0.55
            Debris:AddItem(branchPart, bLifetime + 0.05)
            Debris:AddItem(bAttach0, bLifetime + 0.05)

            ts:Create(branchBeam, TweenInfo.new(bLifetime, Enum.EasingStyle.Expo, Enum.EasingDirection.Out), {
                Transparency = NumberSequence.new(1),
                Width0 = 0,
                Width1 = 0
            }):Play()
        end)
    end
end

local function StartGroundArcs()
    if groundArcActive then return end
    groundArcActive = true

    -- Build available source parts list (only parts that actually exist on this character)
    local arcPartsList = {}
    for _, name in ipairs(GROUND_ARC_SOURCE_PARTS) do
        local p = CharacterRef:FindFirstChild(name)
        if p and p:IsA("BasePart") then
            table.insert(arcPartsList, p)
        end
    end
    if #arcPartsList == 0 then
        table.insert(arcPartsList, HRP)
    end

    groundArcThread = task.spawn(function()
        while groundArcActive and CharacterRef and CharacterRef.Parent do
            local isMoving = h and h.Parent and h.MoveDirection.Magnitude > 0
            local speed = h and h.Parent and h.WalkSpeed or 0

            if speed > 30 then
                local lightColor = GetLightningColor()

                if isMoving then
                    -- Running: fire 1-4 simultaneous arcs from random body parts
                    -- G_Optimized: always 1 arc; Normal: 1-3 arcs
                    local numArcs = _G.G_Optimized and 1 or math.random(1, 3)
                    for i = 1, numArcs do
                        local part = arcPartsList[math.random(1, #arcPartsList)]
                        -- Each arc is spawned independently so they don't block each other
                        task.spawn(FireGroundArc, part, lightColor)
                    end
                else
                    -- Standing still at high speed: occasional single torso arc
                    -- Simulates the "idle lightning crackling" seen in DCEU scenes
                    if math.random() < 0.35 then
                        task.spawn(FireGroundArc, HRP, lightColor)
                    end
                end
            end

            -- Arc interval scales with speed:
            -- At full speed (375): fires every ~30-70ms = ~15-30 arcs/second
            -- At low speed (100): fires every ~60-110ms = ~9-16 arcs/second
            -- G_Optimized: fires every ~80-150ms regardless of speed
            local interval
            if _G.G_Optimized then
                interval = math.random(8, 15) / 100
            else
                local speedFactor = math.clamp(speed / 375, 0, 1)
                local minMs = math.floor(3 + (1 - speedFactor) * 4)

-- ================================================================
-- MASTER DISABLE FUNCTION
-- Called by all stop paths to guarantee a clean shutdown every time.
-- Fixes the "doesn't properly disable" bug: all trail tables and
-- all dynamic systems are stopped in a single deterministic sequence.
-- ================================================================
local function MasterDisableAll()
    -- 1. Stop dynamic systems FIRST (they may be enabling trails)
    StopLightningFlicker()
    StopGroundArcs()

    -- 2. Stop the particle effect system
    StopEffect()

    -- 3. Disable ALL trail tables explicitly
    V88()   -- t trails
    V90()   -- tlist trails
    DisableCoolStuff()   -- allTrails
    DisableLightningGlobal()   -- particle emitter on HRP

    -- 4. Stop running connections
    if runningConnection then
        runningConnection:Disconnect()
        runningConnection = nil
    end
    if heartbeatConnection then
        heartbeatConnection:Disconnect()
        heartbeatConnection = nil
    end

    -- 5. Disable the enableAll system (TrailsPerPart + animations + sounds)
    disableRunEvents()
    disableAll()

    -- 6. Reset flags
    spd = false
    playerData.spd = false
end

local colorUpdateThread = task.spawn(function()
    while colorUpdateRunning and player and player.Parent and CharacterRef and CharacterRef.Parent do
        local success, err = pcall(function()
            local cs = nil
            local spot = nil

            if speedsterType == "RF" then
                cs = rfCS2
                spot = Color3.fromRGB(255,0,0)
            elseif speedsterType == "KidFlash" then
                cs = kidCS2
                spot = Color3.fromRGB(255,255,0)
            elseif speedsterType == "Zoom" then
                cs = zoomCS2
                spot = Color3.fromRGB(124,194,255)
            elseif speedsterType == "GodOfSpeed" then
                cs = godCS2
                spot = Color3.fromRGB(255,220,0)
            elseif speedsterType == "NWA" then
                cs = nwaCS2
                spot = Color3.fromRGB(200,100,255)
            elseif speedsterType == "Regular" then
                cs = regularCS2
                spot = Color3.fromRGB(255,59,33)
            else
                cs = regularCS2
                spot = Color3.fromRGB(255,59,33)
            end

            if cs and CharacterRef and CharacterRef.Parent then
                mk()

                for _,n in pairs(bNames) do
                    local pr = CharacterRef:FindFirstChild(n)
                    if pr and pr.Parent then
                        for _,tr in pairs(pr:GetChildren()) do
                            if tr:IsA("Trail") and tr.Parent then
                                tr.Color = cs
                            end
                        end
                    end
                end

                for _,v in pairs(CharacterRef:GetDescendants()) do
                    if v:IsA("ParticleEmitter") and v.Parent then
                        local cur = v
                        local g = false
                        while cur do
                            if cur == ig then
                                g = true
                                break
                            end
                            cur = cur.Parent
                        end

                        if v.Texture == "rbxassetid://9731596776" then
                            g = true
                        end
                        if v.Texture == "rbxassetid://10583327833" then
                            v.Color = cs
                            g = true
                        end

                        if not g then
                            v.Color = cs
                        end
                    elseif v:IsA("SpotLight") and v.Parent then
                        v.Color = spot
                        v.Enabled = true
                    end
                end
            end
        end)
        
        if not success then
            warn("Color update error for " .. player.Name .. ": " .. tostring(err))
        end

        -- G_Optimized: update color less frequently (0.5s vs 0.1s)
        -- Color rarely changes so 0.1s was waste; 0.5s is imperceptible
        task.wait(_G.G_Optimized and 0.5 or 0.1)
    end
    colorUpdateRunning = false
end)

end
